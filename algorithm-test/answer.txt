function findMaxInMin(arr, groupLength) {
    if (!arr || groupLength <= 0 || groupLength > arr.length) {
        return null;
    }

    let maxOfMinimums = -Infinity;

    for (let startIndex = 0; startIndex <= arr.length - groupLength; startIndex++) {
        let currentMinimum = arr[startIndex];

        for (let offset = 1; offset < groupLength; offset++) {
            let currentValue = arr[startIndex + offset];
            if (currentValue < currentMinimum) {
                currentMinimum = currentValue;
            }
        }

        if (currentMinimum > maxOfMinimums) {
            maxOfMinimums = currentMinimum;
        }
    }

    return maxOfMinimums;
}

// Contoh Pengujian
const arr = [8, 1, 6, 3, 66, 1, 8, 5];
const groupLength = 3;

console.log("Hasil:", findMaxInMin(arr, groupLength)); // Output yang diharapkan: 3

Penjelasan nya:
arr = [8, 1, 6, 3, 66, 1, 8, 5]
groupLength = 3

Rincian langkah demi langkah dari setiap kelompoknya
[8,1,6]   -> 1 // Angka terkecil (minimum) di dalam kelompok
[1,6,3]   -> 1 // Angka terkecil (minimum) di dalam kelompok
[6,3,66]  -> 3 // Angka terkecil (minimum) di dalam kelompok
[3,66,1]  -> 1 // Angka terkecil (minimum) di dalam kelompok
[66,1,8]  -> 1 // Angka terkecil (minimum) di dalam kelompok
[1,8,5]   -> 1 // Angka terkecil (minimum) di dalam kelompok

Dari seluruh nilai minimum yang terkumpul dari setiap kelompok, yaitu [1, 1, 3, 1, 1, 1],
program kemudian mencari nilai yang paling besar (maximum of minimums).

Nilai terbesar dari kumpulan angka minimum tersebut adalah 3. 
Itulah sebabnya fungsi ini mengembalikan hasil akhir 3.

Complexity:
- Time: O((n - k + 1) * k)
- Extra space: O(1)
